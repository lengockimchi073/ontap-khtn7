function shuffleArray(array) {

  const arr = [...array];

  for (let i = arr.length - 1; i > 0; i--) {

    const j = Math.floor(Math.random() * (i + 1));

    [arr[i], arr[j]] = [arr[j], arr[i]];

  }

  return arr;

}

 

function startQuiz(secKey, mode) {

  currentSectionKey = secKey;

  currentPage = 1;

 

  if (mode === 'new') {

    userAnswers = {};

    quizSubmitted = false;

    // Trộn câu hỏi độc lập, giữ nguyên cấu trúc đáp án A-D bên trong mỗi câu

    currentShuffledQuestions = shuffleArray(rawQuestions);

    saveCurrentSession();

  } else {

    loadSavedSession();

    if (!currentShuffledQuestions || currentShuffledQuestions.length === 0) {

      currentShuffledQuestions = shuffleArray(QUIZ_DATA[currentGrade].sections[secKey].questions);

      saveCurrentSession();

    }

  }

 

  document.getElementById("view-sections").style.display = "none";

  document.getElementById("view-quiz").style.display = "block";

  document.getElementById("select-paging").value = currentPaging;

  document.getElementById("quiz-title-badge").textContent = QUIZ_DATA[currentGrade].sections[secKey].title;

}

function updateProgressDisplay() {

  const total = currentShuffledQuestions.length;

  let answered = 0;

  if (currentSectionKey === 'mcq') {

    answered = Object.keys(userAnswers).filter(k => userAnswers[k] !== undefined).length;

  } else if (currentSectionKey === 'tf') {

    let count = 0;

    currentShuffledQuestions.forEach(q => {

      q.items.forEach(it => {

        if (userAnswers[it.id] !== undefined) count++;

      });

    });

    answered = count;

    const totalItems = currentShuffledQuestions.reduce((acc, q) => acc + q.items.length, 0);

    document.getElementById("quiz-progress").textContent = `Đã trả lời: ${answered}/${totalItems} nhận định`;

    return;

  } else {

    answered = Object.keys(userAnswers).filter(k => (userAnswers[k] || '').trim().length > 0).length;

  }

  document.getElementById("quiz-progress").textContent = `Đã trả lời: ${answered}/${total} câu`;

}

        }

      if (allDone) btn.classList.add("done");

function renderQuiz() {

  updateProgressDisplay();

  renderNavButtons();

 

  const container = document.getElementById("questions-container");

  container.innerHTML = "";

 

  const total = currentShuffledQuestions.length;

  let startIndex = 0;

  let endIndex = total;

 

  if (currentPaging !== "all") {

    const pageSize = parseInt(currentPaging, 10);

    const maxPages = Math.ceil(total / pageSize);

    if (currentPage > maxPages) currentPage = maxPages;

    if (currentPage < 1) currentPage = 1;

 

    startIndex = (currentPage - 1) * pageSize;

    endIndex = Math.min(startIndex + pageSize, total);

 

    document.getElementById("btn-prev").style.visibility = (currentPage > 1) ? "visible" : "hidden";

    document.getElementById("btn-next").style.visibility = (currentPage < maxPages) ? "visible" : "hidden";

  } else {

    document.getElementById("btn-prev").style.visibility = "hidden";

    document.getElementById("btn-next").style.visibility = "hidden";

  }

 

  const slice = currentShuffledQuestions.slice(startIndex, endIndex);

 

  slice.forEach((q, sliceIdx) => {

    const realIdx = startIndex + sliceIdx + 1;

    const qDiv = document.createElement("div");

    qDiv.className = "q-item";

    qDiv.id = `q_block_${q.id}`;

 

    if (currentSectionKey === 'mcq') {

      renderMcqItem(qDiv, q, realIdx);

    } else if (currentSectionKey === 'tf') {

      renderTfItem(qDiv, q, realIdx);

    } else {

      renderEssayItem(qDiv, q, realIdx);

    }

 

    container.appendChild(qDiv);

  });

 

  const submitBtn = document.getElementById("btn-submit");

  if (quizSubmitted) {

    submitBtn.textContent = "Làm lại phần này";

    submitBtn.onclick = () => resetCurrentQuiz();

  } else {

    submitBtn.textContent = "Nộp bài kiểm tra";

    submitBtn.onclick = () => confirmSubmitQuiz();

  }

}

  container.innerHTML = `

          <button class="btn btn-sm ${selfEval === 'hieu_mot_phan' ? 'btn-primary' : 'btn-outline'}" onclick="saveEssayEval('${q.id}', 'hieu_mot_phan')">Đã hiểu một phần</button>

      </div>

}

 

function saveEssayText(qId, val) {

  userAnswers[qId] = val;

  saveCurrentSession();

  updateProgressDisplay();

  renderNavButtons();

}

 

function toggleCompareEssay(qId) {

  userAnswers[`compared_${qId}`] = !userAnswers[`compared_${qId}`];

  saveCurrentSession();

  renderQuiz();

}

 

function toggleChecklistItem(qId, idx, checked) {

  userAnswers[`check_${qId}_${idx}`] = checked;

  saveCurrentSession();

}

 

function showMissingChecklist(qId) {

  const q = QUIZ_DATA[currentGrade].sections.essay.questions.find(item => item.id === qId);

  const missing = [];

  q.checklist.forEach((item, idx) => {

    if (!userAnswers[`check_${qId}_${idx}`]) {

      missing.push(item);

    }

  });

 

  const target = document.getElementById(`missing_box_${qId}`);

  if (missing.length === 0) {

    target.innerHTML = `<span style="color:var(--success); font-weight:bold;">✓ Tuyệt vời! Bạn đã hoàn thành trọn vẹn tất cả các ý then chốt.</span>`;

  } else {

    target.innerHTML = `<strong>Ý cần bổ sung thêm:</strong><ul style="margin-left:1.25rem; margin-top:0.25rem;">${missing.map(m => `<li>${m}</li>`).join('')}</ul>`;

  }

}

 

function saveEssayEval(qId, level) {

  saveCurrentSession();

  renderQuiz();

}

function confirmSubmitQuiz() {

  const total = currentShuffledQuestions.length;

  let unanswered = 0;

 

  if (currentSectionKey === 'mcq') {

    unanswered = currentShuffledQuestions.filter(q => userAnswers[q.id] === undefined).length;

  } else if (currentSectionKey === 'tf') {

    let un = 0;

    currentShuffledQuestions.forEach(q => {

      q.items.forEach(it => {

        if (userAnswers[it.id] === undefined) un++;

      });

    });

    unanswered = un;

  } else {

    unanswered = currentShuffledQuestions.filter(q => !(userAnswers[q.id] || '').trim()).length;

  }

 

  const msg = (unanswered > 0)

    ? `Bạn còn <strong>${unanswered}</strong> mục chưa hoàn thành. Bạn có muốn tiếp tục nộp bài không?`

    : `Bạn đã trả lời đầy đủ tất cả các câu. Hãy sẵn sàng nộp bài để xem phản hồi chi tiết!`;

 

  document.getElementById("modal-submit-msg").innerHTML = msg;

  document.getElementById("modal-submit").style.display = "flex";

}
